// Temporary integration test (crates/cli/tests/zz_long.rs): a long chat until the checkpoint copy no longer fits.
// Run: MODEL=~/models/Qwen3.5-4B-Q8 CTX=15360 TURNS=10 cargo test --release -p cli --test zz_long -- --nocapture
//! TEMPORARY measurement harness for the performance investigation; deleted afterwards.
use std::path::PathBuf;
use std::process::Command;
use std::time::Instant;

use cli::dialog::{Dialog, Talk};
use cli::load::{ChatPrecision, load_chat};
use cli::session::{Session, Settings};
use cli::setup::precision_for_pick;
use cli::signal::Interrupt;
use cli::style::Style;
use model::BackendChoice;
use runtime::SequenceCache;

fn mine() -> String {
    let pid = std::process::id().to_string();
    let out = Command::new("nvidia-smi").args(["--query-compute-apps=pid,used_memory", "--format=csv,noheader,nounits"]).output().unwrap();
    String::from_utf8_lossy(&out.stdout).lines().find(|l| l.starts_with(&pid)).map(|l| l.split(", ").nth(1).unwrap_or("?").to_string() + " MiB").unwrap_or_else(|| "?".into())
}

#[test]
fn long_chat() {
    let model = PathBuf::from(std::env::var("MODEL").unwrap());
    let ctx: usize = std::env::var("CTX").unwrap().parse().unwrap();
    let turns: usize = std::env::var("TURNS").unwrap().parse().unwrap();
    let interrupt = Interrupt::unhooked();
    let asked = ChatPrecision { weights: None, kv: None };
    let loaded = load_chat(&model, (Some(ctx), ctx), BackendChoice::Cuda, precision_for_pick(asked, &model), &Style::plain(), (false, false)).unwrap();
    let settings = Settings { max_tokens: 8, temperature: 0.0, top_k: 0, top_p: 1.0, seed: Some(1) };
    let mut s = Session::new(loaded, settings, Style::plain()).unwrap();
    println!("LOADED ctx {ctx}: process VRAM {}", mine());
    let mut dialog = Dialog::new(&s, Talk::Chat, std::env::current_dir().unwrap(), None).unwrap();
    let filler = |n: usize| format!("Note {n}: the lighthouse keeper climbed the spiral stairs each evening, trimmed the lamp and watched the ships. ").repeat(60);
    for n in 0..turns {
        let started = Instant::now();
        let r = dialog.chat.ask(&mut s, &filler(n), &interrupt, &mut Vec::new());
        match r {
            Ok(reply) => {
                let o = &reply.outcome;
                println!("TURN {:2}: prompt {:6} reused {:6} read {:5} wall {:6.1}s cache {:6} tok, live KV {:?} | process VRAM {}", n + 1, o.prompt_tokens, o.reused, o.prompt_tokens - o.reused, started.elapsed().as_secs_f64(), s.kv().used(), s.kv().kv_held().map(|(p, b)| format!("{p} pages {} MiB", b / 1048576)), mine());
            }
            Err(e) => {
                println!("TURN {:2}: error {e}", n + 1);
                break;
            }
        }
    }
}
