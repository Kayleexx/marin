// Run: MODEL=~/models/Qwen3.5-4B-Q8 N=9856 MARIN_REQUIRE_CUDA=1 cargo test --release -p model --test zz_mem -- --nocapture
//! TEMPORARY memory trace for the performance investigation; deleted afterwards.
use model::{Backend, Engine, QuantBits, QuantScheme};
use runtime::CausalLm;

#[test]
fn trace() {
    let dir = loader::ModelDir::open(std::path::Path::new(&std::env::var("MODEL").unwrap())).unwrap();
    let gpu = model::Gpu::new().unwrap();
    let scheme = QuantScheme::new(QuantBits::Int8, model::DEFAULT_GROUP).unwrap();
    let engine = Engine::load(&dir, Backend::Cuda(Box::new(gpu)), Some(scheme), |_, _| {}).unwrap();
    let used = || {
        let (free, total) = engine.gpu_memory().unwrap();
        (total - free) as f64 / 1048576.0
    };
    let n: usize = std::env::var("N").unwrap().parse().unwrap();
    let base = used();
    let step = |label: &str, cache: Option<&model::EngineCache>| {
        let held = cache.and_then(|c| c.kv_held()).map(|(p, b)| format!("live {p} pages {} MiB", b / 1048576)).unwrap_or_default();
        println!("MEM {label:<40} used {:8.0} MiB  (+{:6.0} vs loaded)  {held}", used(), used() - base);
    };
    step("model loaded", None);
    let mut cache = engine.new_cache(15360).unwrap();
    step("new_cache(15360)", Some(&cache));
    for chunk in (0..n).collect::<Vec<_>>().chunks(64) {
        let ids: Vec<u32> = chunk.iter().map(|i| 1000 + (*i as u32 % 500)).collect();
        engine.forward(&ids, &mut cache).unwrap();
    }
    step(&format!("after prefill {n} tokens (64-slices)"), Some(&cache));
    let (free_before, total) = engine.gpu_memory().unwrap();
    let copy = engine.session_checkpoint(&cache).unwrap();
    println!("MEM checkpoint decision at {n} tokens: free {:.0} MiB of {:.0} MiB, live KV+state {} MiB -> {}", free_before as f64 / 1048576.0, total as f64 / 1048576.0, cache.kv_held().map_or(0, |(_, b)| b / 1048576), if copy.is_some() { "copy made" } else { "REFUSED (plain fallback)" });
    step("after session_checkpoint", Some(&cache));
    let Some(copy) = copy else { return };
    let copy = Some(copy);
    for t in 0..8u32 {
        engine.forward(&[1100 + t], &mut cache).unwrap();
    }
    step("after 8 decode tokens on live cache", Some(&cache));
    let mut copy = copy.unwrap();
    drop(cache);
    step("after dropping the live cache", Some(&copy));
    engine.forward(&[1], &mut copy).unwrap();
    step("after one forward on the copy (sync)", Some(&copy));
    drop(copy);
    step("after dropping the copy", None);
}
