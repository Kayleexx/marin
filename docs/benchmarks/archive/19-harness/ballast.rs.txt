#![allow(dead_code)]
//! TEMPORARY: holds GPU memory until a file disappears, to test low-VRAM behaviour.
mod common;
#[allow(dead_code)]
fn unused() {}
#[test]
fn hold() -> Result<(), Box<dyn std::error::Error>> {
    let Some(gpu) = common::gpu()? else { return Ok(()) };
    let mib: usize = std::env::var("MIB")?.parse()?;
    let flag = std::env::var("FLAG")?;
    let _held = gpu.zeros(mib * 262_144)?;
    gpu.synchronize()?;
    println!("BALLAST held {mib} MiB");
    while std::path::Path::new(&flag).exists() {
        std::thread::sleep(std::time::Duration::from_millis(200));
    }
    Ok(())
}
