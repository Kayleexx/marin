use model::qwen35::Qwen35Gpu;
use model::{QuantBits, QuantScheme};
use runtime::{CausalLm, SequenceCache};
use std::sync::atomic::{AtomicBool, AtomicU64, Ordering};

#[test]
fn memory() -> Result<(), Box<dyn std::error::Error>> {
    let Some(dir) = std::env::var_os("MARIN_QWEN35_Q8_DIR") else {
        return Ok(());
    };
    let packed = loader::ModelDir::open(std::path::Path::new(&dir))?;
    let scheme = QuantScheme::new(QuantBits::Int8, model::DEFAULT_GROUP)?;
    let m = Qwen35Gpu::load(&packed, model::Gpu::new()?, Some(scheme), |_, _| {})?;
    let free = || m.gpu().info().map(|i| i.free_bytes).unwrap_or(0);
    let prompt: Vec<u32> = (100..140).collect();
    {
        let (mut a, mut b) = (m.new_cache(512)?, m.new_cache(512)?);
        m.forward(&prompt, &mut a)?;
        m.forward(&prompt, &mut b)?;
        for t in 300..312u32 {
            let x = m.forward(&[t], &mut a)?;
            let y = m.forward(&[t], &mut b)?;
            assert!(x == y);
        }
        let n = b.used();
        b.truncate(n - 1)?;
        let replayed = m.forward(&[311], &mut b)?;
        let mut c = m.new_cache(512)?;
        m.forward(&prompt, &mut c)?;
        for t in 300..311u32 { m.forward(&[t], &mut c)?; }
        let incremental = m.forward(&[311], &mut c)?;
        let diff = replayed.iter().zip(&incremental).map(|(a, b)| (a - b).abs()).fold(0f32, f32::max);
        let top = incremental.iter().fold(0f32, |m, x| m.max(x.abs()));
        let am = |v: &[f32]| v.iter().enumerate().max_by(|a, b| a.1.total_cmp(b.1)).map(|x| x.0);
        eprintln!("replay vs incremental: identical {} max abs diff {diff:e} (max logit {top}) same argmax {}", replayed == incremental, am(&replayed) == am(&incremental));
    }
    for b in [1usize, 2, 3] {
        let mut caches = Vec::new();
        for _ in 0..b {
            let mut c = m.new_cache(2048)?;
            m.forward(&prompt, &mut c)?;
            caches.push(c);
        }
        let toks = vec![300u32; b];
        for mode in ["batch"] {
            let base = free();
            let low = AtomicU64::new(base as u64);
            let stop = AtomicBool::new(false);
            std::thread::scope(|s| -> Result<(), Box<dyn std::error::Error>> {
                s.spawn(|| {
                    let _ = m.gpu().bind_to_thread();
                    while !stop.load(Ordering::Relaxed) {
                        low.fetch_min(free() as u64, Ordering::Relaxed);
                    }
                });
                for step in 0..150 {
                    if step % 50 == 0 {
                        eprint!("{} ", free() >> 20);
                    }
                    if mode == "batch" {
                        let mut r: Vec<_> = caches.iter_mut().collect();
                        m.forward_batch(&toks, &mut r)?;
                    } else {
                        for c in caches.iter_mut() {
                            m.forward(&[300], c)?;
                        }
                    }
                }
                stop.store(true, Ordering::Relaxed);
                Ok(())
            })?;
            let after = free();
            eprintln!(
                "B={b} {mode}: peak +{:.1} MiB over resting, resting after {:+.1} MiB",
                (base as f64 - low.load(Ordering::Relaxed) as f64) / 1048576.0,
                (base as f64 - after as f64) / 1048576.0
            );
        }
    }
    Ok(())
}
